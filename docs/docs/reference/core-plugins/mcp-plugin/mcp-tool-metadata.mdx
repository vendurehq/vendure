Describes a single MCP tool. You attach this to a class with the [McpTool](/reference/core-plugins/mcp-plugin/mcp-tool#mcptool)decorator. The MCP server finds those classes on startup and exposes each one as a
callable tool.

```ts title="Signature"
interface McpToolMetadata {

    title?: string;
    description: string;
    keywords?: string[];
    toolset: McpToolset;
    permissions?: Permission[];
    behavior?: McpToolBehavior;
    inputSchema?: McpToolSchema;
    outputSchema?: McpToolSchema;
}
```
